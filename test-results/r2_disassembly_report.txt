# Prototype Concept

0x140ab1c98  lea rdx, [rsp+20]      ; rdx = pointer to HUD matrix
0x140ab1c9d  call fcn.141838500     ; apply matrix by handle

using ApplyMatrixFn = void (*)(uint32_t handle, float* matrix);

static ApplyMatrixFn OriginalApplyMatrix = nullptr;

static void HookHudMatrix(uint32_t handle, float* matrix)
{
    if (matrix)
    {
        matrix[12] += 0.25f; // try row-3 translation
        // try matrix[13] or matrix[14] if this axis is not obvious
    }

    OriginalApplyMatrix(handle, matrix);
}

uintptr_t base = reinterpret_cast<uintptr_t>(GetModuleHandleA(nullptr));

OriginalApplyMatrix =
    reinterpret_cast<ApplyMatrixFn>(base + 0x1838500);

PatchRelativeCall(
    base + 0xAB1C9D,
    reinterpret_cast<uintptr_t>(&HookHudMatrix)
);

static bool PatchRelativeCall(uintptr_t callAddress, uintptr_t hookAddress)
{
    intptr_t rel = static_cast<intptr_t>(hookAddress) -
        static_cast<intptr_t>(callAddress + 5);

    if (rel < INT32_MIN || rel > INT32_MAX)
        return false;

    DWORD oldProtect = 0;
    if (!VirtualProtect(reinterpret_cast<void*>(callAddress), 5, PAGE_EXECUTE_READWRITE, &oldProtect))
        return false;

    unsigned char* p = reinterpret_cast<unsigned char*>(callAddress);
    p[0] = 0xE8; // call rel32
    *reinterpret_cast<int32_t*>(p + 1) = static_cast<int32_t>(rel);

    DWORD ignored = 0;
    VirtualProtect(reinterpret_cast<void*>(callAddress), 5, oldProtect, &ignored);
    FlushInstructionCache(GetCurrentProcess(), reinterpret_cast<void*>(callAddress), 5);
    return true;
}
